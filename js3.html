Objectif : Maîtriser les fonctions (déclarations, expressions, arrow functions, portée, closures).

Formation 30 jours — HTML / CSS / JavaScript

1. Déclarations de fonctions

Function Declaration

// Déclaration de fonction
function additionner(a, b) {
    return a + b;
}

const resultat = additionner(5, 3);
console.log(resultat); // 8

Function Expression

// Expression de fonction
const multiplier = function(a, b) {
    return a * b;
};

console.log(multiplier(5, 3)); // 15

Arrow Function

// Arrow function (retour implicite)
const carre = x => x * x;
console.log(carre(5)); // 25

// Arrow function avec corps
const additionner = (a, b) => {
    return a + b;
};
console.log(additionner(5, 3)); // 8

💡 Différences :

  • Function Declaration : hoisted, peut être appelée avant déclaration
  • Function Expression : non hoisted, assignée à une variable
  • Arrow Function : syntaxe concise, pas de this propre, pas d'arguments

2. Paramètres

Paramètres par défaut

function saluer(nom = "inconnu", age = 0) {
    console.log(`Bonjour ${nom}, ${age} ans.`);
}

saluer(); // "Bonjour inconnu, 0 ans."
saluer("Jean"); // "Bonjour Jean, 0 ans."
saluer("Jean", 25); // "Bonjour Jean, 25 ans."

Paramètres rest

// Regroupe les arguments restants
function somme(...nombres) {
    return nombres.reduce((acc, n) => acc + n, 0);
}

console.log(somme(1, 2, 3)); // 6
console.log(somme(1, 2, 3, 4, 5)); // 15

Destructuration des paramètres

function afficherPersonne({ nom, age, ville = "inconnue" }) {
    console.log(`Nom: ${nom}, Âge: ${age}, Ville: ${ville}`);
}

const personne = { nom: "Jean", age: 25 };
afficherPersonne(personne); // "Nom: Jean, Âge: 25, Ville: inconnue"

3. Portée et Closures

Portée de fonction

function testPortee() {
    const local = "Je suis locale";
    console.log(local); // Accessible
}

testPortee();
// console.log(local); // Erreur

Closure (fermeture)

function creerCompteur() {
    let compteur = 0;
    return function() {
        compteur++;
        return compteur;
    };
}

const compteur1 = creerCompteur();
console.log(compteur1()); // 1
console.log(compteur1()); // 2
console.log(compteur1()); // 3

const compteur2 = creerCompteur();
console.log(compteur2()); // 1 (indépendant)

💡 Closure :

  • Une fonction qui se souvient de son environnement
  • Permet de créer des variables privées
  • Utilisée dans les compteurs, les fabriques de fonctions
